\begin{frame}[allowframebreaks]{Sampling}
    \begin{itemize}
        \item Abtastung stetiger Signale $\implies$ Folge diskreter Amplitudenwerte
    \end{itemize}
    \begin{figure}
        \centering
        \includegraphics[width=0.6\textwidth]{Graphics/Diskretes_signal_lyons.png}
        \caption{
            Horiz. Achse: Zeitbereich (n-ter Samplewert) 
            Vert. Achse: Amplitude
            \cite[Kap. 2]{lyons2011}
            }
    \end{figure}
    \begin{itemize}
        \item{Berechnung von Signalwerten}:
    \end{itemize}
    \begin{equation}
        x[n] = \sin(2\pi \cdot f_0 \cdot n \cdot t_s)
    \end{equation}
    Wobei: 
    \begin{itemize}
        \item{$2\pi$}: Eine komplette Schwingung (360° auf dem Einheitskreis)
        \item{$f_0$}: Frequenz des Signals in Hz
        \item{$n$}: Sample-Index
        \item{$f_s$}: Sampling-Intervall (Zeit, die zwischen zwei Messungen vergeht)
    \end{itemize}
    
    
        
    
\end{frame}
