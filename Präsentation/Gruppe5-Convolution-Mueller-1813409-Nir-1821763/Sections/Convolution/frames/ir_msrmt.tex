\begin{frame}[allowframebreaks]{Messung von Impulsantworten}

    Verschiedene Methoden:
    \linebreak
    \begin{itemize}
        \item Minimum Length Sequence (MLS): digital erzeugtes Rauschen mit praktischer Eigenschaft
        \item Time Delay Spectroscopy (TDS): linearer Sinus-Sweep
        \item Logarithmic/Exponential Sine Sweep (LSS/ESS): logarithmischer/exponentieller Sinus-Sweep
    \end{itemize}

    \framebreak

    LSS/ESS-Methode für Convolution Reverb am besten geeignet.
    \linebreak
    \begin{itemize}
        \item Kann mit leichten Nonlinearen Eigenschaften umgehen
        \item Kein Rauschen durch Zeitvariante Eigenschaften
        \item Deconvolution durch Convolution mit inversem Sinus-Sweep
    \end{itemize}
    
\end{frame}
